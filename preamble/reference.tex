%% Set up cross-referencing and hyperlinks.
\usepackage{hyperref}
\usepackage{zref-titleref}
\usepackage{zref-clever}
\zcsetup{noabbrev}
\hypersetup{
  linktoc = all,
  colorlinks = true,
  citecolor = black,
  linkcolor = black,
  urlcolor = [rgb]{0.82,0.25,0.08},
  pdfborder = {0 0 0},
  pdflinkmargin = 1.0pt,
  pdfdisplaydoctitle,
  pdftitle = {\homeworksubject\ Homework \homeworkindex},
  pdfauthor = {\homeworkstudent},
}

%% Set up cross-referencing style.
\usepackage[most]{tcolorbox}

% colors
\definecolor{UrlColor}{RGB}{255,127,100}

% styles for boxes
\tcbset{
  link/.style={
    nobeforeafter, box align=base, boxsep=0em, before upper={\vphantom{]}},
    left=0.15em, right=0.15em, top=0.25ex, bottom=0.05ex,
    arc=0.1em, rounded corners=all, boxrule=0.3mm,
    grow sidewards by=0.05em, enlarge bottom by=-0.05ex, enlarge top by=-0.4ex,
  }
}
\tcbset{
  refbox/.style ={colframe=black!30, colback=black!3, fontupper=\sffamily},
  pagebox/.style={refbox, boxsep=0.1ex, bottom=0.25ex, fontupper=\ttfamily},
  hrefbox/.style={colframe=black!10!UrlColor!70, colback=UrlColor!15},
}
\NewTCBox{\refbox} {!O{}}{link,refbox, #1}
\NewTCBox{\pagebox}{!O{}}{link,pagebox,#1}
\NewTCBox{\hrefbox}{!O{}}{link,hrefbox,#1}

% save original commands
\NewCommandCopy{\hrefOrig}{\href}
\NewCommandCopy{\urlOrig}{\url}

% link command with boxes
\RenewDocumentCommand{\href}{O{} m m}{\hrefOrig{#2}{\hrefbox[#1]{#3}}}
\RenewDocumentCommand{\url}{O{} m}{\href[#1]{#2}{\nolinkurl{#2}}}

% cross-reference command with boxes
\DeclareDocumentCommand{\Cref}{s m}{%
  {%
    \normalfont
    \IfBooleanTF{#1}
      {% multiple references
        \small\refbox{\zcref[S]{#2}}%
      }
      {% single reference
        \hyperref[#2]{%
            \small\refbox{%
              \zcref*[S,noref]{#2}~\zcref*[S,noname]{#2}%
            }% 
          }%
      }%
  }%
}
\DeclareDocumentCommand{\cref}{s m}{%
  {%
    \normalfont
    \IfBooleanTF{#1}
      {% multiple references
        \small\refbox{\zcref{#2}}%
      }
      {% single reference
        \hyperref[#2]{%
          \small\refbox{%
            \zcref*[noref]{#2}~\zcref*[noname]{#2}%
          }% 
        }%
      }%
  }%
}
